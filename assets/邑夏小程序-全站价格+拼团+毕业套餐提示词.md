### 毕业派对套餐的拼团交互

当产品ID为 `pkg_graduation_party` 时，底部操作区显示拼团档位选择：

```tsx
{productId === 'pkg_graduation_party' && (
  <View className="pintuan-tiers-bar">
    <View className="pintuan-tier-btn" onClick={() => handlePintuanTier(10)}>
      <Text className="pintuan-tier-price">¥69/人</Text>
      <Text className="pintuan-tier-desc">10人团</Text>
    </View>
    <View className="pintuan-tier-btn pintuan-tier-hot" onClick={() => handlePintuanTier(20)}>
      <Text className="pintuan-tier-price">¥29.9/人</Text>

  </View>

```

拼团发起逻辑：

```typescript
const handlePintuanTier = (minPeople: number) => {
  const priceLabel = minPeople === 10 ? '¥69' : '¥29.9'
  Taro.showModal({
    title: '发起拼团',
    content: `${minPeople}人成团，每人${priceLabel}。分享给好友即可参团！`,
    confirmText: '去分享',
    success: (res) => {
      if (res.confirm) {
        Taro.showShareMenu({
          withShareTicket: true,
          menus: ['shareAppMessage', 'shareTimeline']
        })
      }
    }
  })
}

// 分享配置
Taro.useShareAppMessage(() => {
  const isPartyPkg = productId === 'pkg_graduation_party'
  return {
    title: isPartyPkg
      ? '邑夏毕业派对套餐拼团！20人团仅¥29.9/人，11瓶畅饮！'
      : `邑夏${productName}拼团，2人成团仅¥15.8！`,
    path: `/pages/product/detail?id=${productId}&pintuan=true`,
    imageUrl: productImage
  }
})
```
