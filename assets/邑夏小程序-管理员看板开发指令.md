## ⛔ 执行铁律（继承v4.0全部铁律，以下为补充）

### 铁律A：精确修改，不重写
- 修改已有文件时，必须先read_file确认当前内容，再用edit_file精确替换
- 禁止用write_file覆盖已有文件
- 新建文件才用write_file

### 铁律B：Mock数据必须真实可渲染
- 所有数字必须有合理值（不能是0或空）
- 辅导员姓名用真实中文姓名
- 产品名用已确认的6款果酒命名（榴红心事/番红暗许/桃心暗动/清苹微醉/葡香暗度/为爱而生）
- 学校名用"青岛农业大学"
- 操作状态必须与数据一致（下架产品标记为off_sale，库存为0时标红等）

### 铁律C：每个操作按钮必须有实际交互
- 上架/下架 → 产品状态切换
- 库存修改 → 弹出输入框，修改后即时刷新显示
- 发布活动 → 弹出表单，提交后出现在活动列表
- 禁止出现"TODO""待对接""占位"等未实现操作

// ====== 产品管理 ======
export type ProductStatus = 'on_sale' | 'off_sale' | 'pre_sale'

export interface AdminProduct {
  id: string
  name: string
  category: 'fruit_wine' | 'grain_wine' | 'nfc_juice' | 'gift_box'
  price: number
  originalPrice: number
  spec: string               // 如 "330ml"
  stock: number              // 库存
  status: ProductStatus
  isRecommended: boolean     // 是否新品推荐
  recommendOrder: number     // 推荐排序（0=不推荐，1=最高优先）
  sales: number              // 月销量
  icon: string               // 产品图标emoji
  ipCharacter?: string       // 关联的器官大人角色名
  description?: string       // 产品描述
  lastStockUpdate: string    // 最后库存更新时间
}

// ====== 辅导员管理 ======
export type CounselorTier = 'junior' | 'senior' | 'gold'

export interface CounselorData {
  id: string
  name: string
  avatar: string
  phone: string
  tier: CounselorTier
  registeredStudents: number
  monthlySales: number
  monthlyCommission: number
  totalCommission: number
  offlineEvents: number
  onlineRate: number          // 线上提成率（0.08-0.15）
  offlineRate: number         // 线下提成率（0.10-0.20）
  status: 'active' | 'inactive'
  joinDate: string
  counselorCode: string       // 专属推广码
}

// ====== 线下活动 ======
export type ActivityStatus = 'draft' | 'published' | 'ongoing' | 'completed' | 'cancelled'

export interface OfflineActivity {
  id: string
  title: string               // 如 "618品酒会"
  date: string                // 活动日期
  time: string                // 时间段 "19:00-22:00"
  location: string            // 地点
  counselorId: string         // 负责辅导员
  counselorName: string
  maxAttendees: number        // 最大人数
  registeredCount: number     // 已报名
  checkedInCount: number      // 已核销到场
  onSiteSales: number         // 现场销售额
  onSiteOrders: number        // 现场订单数
  status: ActivityStatus
  description: string         // 活动描述
  ticketPrice: number         // 门票价格（0=免费）
  checkInCode: string         // 核销二维码标识
}

// ====== IP管理 ======
export interface IPCharacter {
  id: string
  organRole: string           // liver/heart/spleen/lung/kidney
  name: string                // 如 "楂楂肝谋士"
  emotion: string             // 情绪标签
  color: string               // 主题色
  emoji: string
  wine: string                // 关联酒名
  catchphrase: string         // 口头禅/宣传语
  description: string         // 角色描述
  isActive: boolean           // 是否启用
  lastUpdated: string
}

// ====== 外部链接 ======
export interface ExternalLink {
  id: string
  title: string
  url: string
  category: 'factory' | 'school' | 'partner'  // 厂家/校方/合作伙伴
  icon: string                // emoji图标
  description: string
  isActive: boolean
  sortOrder: number
}

// ====== 消费者画像 ======
export interface UserProfileData {
  totalUsers: number
  activeUsers: number
  newUsersThisMonth: number
  schoolDistribution: { name: string; count: number; percent: number }[]
  tastePreference: { name: string; count: number; percent: number }[]
  consumptionLevel: { level: string; label: string; count: number; percent: number }[]
}

// ====== 看板概览 ======
export interface DashboardOverview {
  todaySales: number
  todayOrders: number
  todayNewUsers: number
  activeCounselors: number
  monthSales: number
  monthOrders: number
  monthNewUsers: number
  lowStockCount: number       // 库存预警数
  pendingActivities: number   // 待处理活动数
  salesTrend: { date: string; sales: number; orders: number }[]
}

interface AdminState {
  // ====== 数据 ======
  dashboard: DashboardOverview
  products: AdminProduct[]
  counselors: CounselorData[]
  activities: OfflineActivity[]
  ipCharacters: IPCharacter[]
  externalLinks: ExternalLink[]
  userProfile: UserProfileData
  
  // ====== 筛选/排序 ======
  productFilter: { category: string | null; status: ProductStatus | null }
  counselorSortBy: 'students' | 'sales' | 'commission'
  activityFilter: ActivityStatus | null
  dateRange: 'today' | 'week' | 'month'
  
  // ====== 产品操作 ======
  toggleProductStatus: (productId: string) => void       // 上架/下架切换
  updateStock: (productId: string, newStock: number) => void  // 修改库存
  toggleRecommend: (productId: string) => void            // 新品推荐开关
  setRecommendOrder: (productId: string, order: number) => void  // 推荐排序
  addProduct: (product: Omit<AdminProduct, 'id'>) => void // 新增产品
  
  // ====== 辅导员操作 ======
  setCounselorSortBy: (sortBy: string) => void
  adjustCommissionRate: (counselorId: string, onlineRate: number, offlineRate: number) => void
  toggleCounselorStatus: (counselorId: string) => void
  upgradeCounselorTier: (counselorId: string, tier: CounselorTier) => void
  
  // ====== 活动操作 ======
  createActivity: (activity: Omit<OfflineActivity, 'id' | 'checkInCode'>) => void
  updateActivityStatus: (activityId: string, status: ActivityStatus) => void
  checkInAttendee: (activityId: string) => void  // 核销+1
  cancelActivity: (activityId: string) => void
  
  // ====== IP操作 ======
  updateIPCharacter: (characterId: string, updates: Partial<IPCharacter>) => void
  toggleIPActive: (characterId: string) => void
  
  // ====== 链接操作 ======
  addExternalLink: (link: Omit<ExternalLink, 'id'>) => void
  updateExternalLink: (linkId: string, updates: Partial<ExternalLink>) => void
  toggleLinkActive: (linkId: string) => void
  removeExternalLink: (linkId: string) => void
  
  // ====== 通用操作 ======
  setDateRange: (range: string) => void
  issueAllUserCoupon: (couponType: string) => void
  getSortedCounselors: () => CounselorData[]
  getFilteredProducts: () => AdminProduct[]
  getFilteredActivities: () => OfflineActivity[]
}
