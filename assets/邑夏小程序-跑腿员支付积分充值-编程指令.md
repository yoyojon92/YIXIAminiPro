/**
 * 支付方式状态管理
 * 支持银行卡/支付宝/微信支付，默认支付方式切换
 */
import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import taroStorage from './taroStorage'

// 支付方式类型
export type PaymentMethodType = 'bank_card' | 'alipay' | 'wechat'

// 银行卡信息
export interface BankCard {
  id: string
  bankName: string       // 银行名称
  bankIcon: string       // 银行图标emoji
  cardNo: string         // 卡号（脱敏）
  cardType: 'debit' | 'credit'  // 储蓄卡/信用卡
  holderName: string     // 持卡人姓名
  isDefault: boolean
}

// 支付宝
export interface AlipayAccount {
  id: string
  account: string        // 支付宝账号（脱敏）
  realName: string       // 实名姓名
  isDefault: boolean
}

// 微信支付（自动绑定，无需手动添加）
export interface WechatPay {
  id: string
  nickname: string
  isDefault: boolean
}

// 支付方式联合
export interface PaymentMethod {
  id: string
  type: PaymentMethodType
  label: string           // 显示名称
  icon: string            // emoji图标
  detail: string          // 脱敏详情
  isDefault: boolean
  data: BankCard | AlipayAccount | WechatPay
}

interface PaymentState {
  methods: PaymentMethod[]
  defaultMethodId: string | null

  // 操作
  addBankCard: (card: Omit<BankCard, 'id' | 'isDefault'>) => void
  removeMethod: (id: string) => void
  setDefault: (id: string) => void
  addAlipay: (account: Omit<AlipayAccount, 'id' | 'isDefault'>) => void
  getDefaultMethod: () => PaymentMethod | undefined
}

const BANK_ICONS: Record<string, string> = {
  '工商银行': '🏦', '建设银行': '🏛️', '农业银行': '🌾',
  '中国银行': '🇨🇳', '招商银行': '🏦', '交通银行': '🚄',
  '邮储银行': '📮', '浦发银行': '🏦', '中信银行': '🏦',
  '光大银行': '☀️', '民生银行': '🏦', '兴业银行': '🏦',
  'default': '💳'
}

const generateId = () => `pm_${Date.now()}_${Math.random().toString(36).substr(2, 6)}`

export const usePaymentStore = create<PaymentState>()(
  persist(
    (set, get) => ({
      methods: [
        {
          id: 'pm_wechat_default',
          type: 'wechat',
          label: '微信支付',
          icon: '💚',
          detail: '微信自动绑定',
          isDefault: true,
          data: { id: 'pm_wechat_default', nickname: '大学生用户', isDefault: true }
        }
      ],
      defaultMethodId: 'pm_wechat_default',

      addBankCard: (card) => {
        const newMethod: PaymentMethod = {
          id: generateId(),
          type: 'bank_card',
          label: card.bankName,
          icon: BANK_ICONS[card.bankName] || BANK_ICONS.default,
          detail: `**** **** **** ${card.cardNo.slice(-4)}`,
          isDefault: false,
          data: { ...card, id: generateId(), isDefault: false }
        }
        set(state => ({ methods: [...state.methods, newMethod] }))
      },

      addAlipay: (account) => {
        const newMethod: PaymentMethod = {
          id: generateId(),
          type: 'alipay',
          label: '支付宝',
          icon: '🔵',
          detail: account.account.replace(/(.{3}).*(@.*)/, '$1****$2'),
          isDefault: false,
          data: { ...account, id: generateId(), isDefault: false }
        }
        set(state => ({ methods: [...state.methods, newMethod] }))
      },

      removeMethod: (id) => {
        set(state => {
          const methods = state.methods.filter(m => m.id !== id)
          if (state.defaultMethodId === id && methods.length > 0) {
            methods[0].isDefault = true
            return { methods, defaultMethodId: methods[0].id }
          }
          return { methods }
        })
      },

      setDefault: (id) => {
        set(state => ({
          defaultMethodId: id,
          methods: state.methods.map(m => ({
            ...m,
            isDefault: m.id === id
          }))
        }))
      },

      getDefaultMethod: () => {
        return get().methods.find(m => m.id === get().defaultMethodId)
      }
    }),
    {
      name: 'yixia-payment-storage',
      partialize: (state) => ({
        methods: state.methods,
        defaultMethodId: state.defaultMethodId
      })
    }
  )
)
```

import { useState } from 'react'
import { View, Text, Input } from '@tarojs/components'

import { Card, CardContent } from '@/components/ui/card'
import { Button } from '@/components/ui/button'
import { Badge } from '@/components/ui/badge'

  CreditCard, Smartphone, CheckCircle

// 银行列表
const BANK_LIST = [
  '工商银行', '建设银行', '农业银行', '中国银行',
  '招商银行', '交通银行', '邮储银行', '浦发银行',
  '中信银行', '光大银行', '民生银行', '兴业银行'
]

export default function Payment() {
  const router = useRouter()
  const { methods, defaultMethodId, setDefault, addBankCard, addAlipay, removeMethod } = usePaymentStore()
  const [showAddSheet, setShowAddSheet] = useState<'none' | 'bank' | 'alipay'>('none')
  
  // 添加银行卡表单
  const [bankForm, setBankForm] = useState({
    bankName: '',
    cardNo: '',
    holderName: '',
    cardType: 'debit' as 'debit' | 'credit'
  })

  // 添加支付宝表单
  const [alipayForm, setAlipayForm] = useState({
    account: '',
    realName: ''
  })

  // 按类型分组
  const groupedMethods: Record<PaymentMethodType, PaymentMethod[]> = {
    wechat: methods.filter(m => m.type === 'wechat'),
    alipay: methods.filter(m => m.type === 'alipay'),
    bank_card: methods.filter(m => m.type === 'bank_card')
  }

  // 提交绑定银行卡
  const handleAddBankCard = () => {
    if (!bankForm.bankName || !bankForm.cardNo || !bankForm.holderName) return
    addBankCard(bankForm)
    setBankForm({ bankName: '', cardNo: '', holderName: '', cardType: 'debit' })
    setShowAddSheet('none')
  }

  // 提交绑定支付宝
  const handleAddAlipay = () => {
    if (!alipayForm.account || !alipayForm.realName) return
    addAlipay(alipayForm)
    setAlipayForm({ account: '', realName: '' })
    setShowAddSheet('none')
  }

  const TYPE_CONFIG: Record<PaymentMethodType, { title: string; icon: any; color: string }> = {
    wechat: { title: '微信支付', icon: Smartphone, color: 'text-green-600' },
    alipay: { title: '支付宝', icon: CreditCard, color: 'text-blue-600' },
    bank_card: { title: '银行卡', icon: CreditCard, color: 'text-purple-600' }
  }

  return (
    <View className="min-h-screen bg-gray-50 pb-safe">
      {/* 当前默认 */}
      <View className="px-4 py-4">
        <Text className="text-sm text-gray-500 mb-2">默认支付方式</Text>
        <Card>
          <CardContent className="p-4">
            <View className="flex items-center gap-3">
              <Text className="text-2xl">
                {methods.find(m => m.id === defaultMethodId)?.icon || '💳'}
              </Text>
              <View className="flex-1">
                <Text className="text-sm font-medium text-gray-900">
                  {methods.find(m => m.id === defaultMethodId)?.label || '未设置'}
                </Text>
                <Text className="text-xs text-gray-500">
                  {methods.find(m => m.id === defaultMethodId)?.detail || ''}
                </Text>
              </View>
              <Badge className="text-xs bg-green-50 text-green-600 border-0">默认</Badge>
            </View>
          </CardContent>
        </Card>
      </View>

      {/* 支付方式列表 */}
      {(Object.keys(TYPE_CONFIG) as PaymentMethodType[]).map(type => {
        const config = TYPE_CONFIG[type]
        const list = groupedMethods[type]
        if (list.length === 0 && type === 'wechat') return null // 微信支付始终存在

        return (
          <View key={type} className="px-4 mb-4">
            <View className="flex items-center justify-between mb-2">
              <Text className="text-sm font-semibold text-gray-900">{config.title}</Text>
              {type !== 'wechat' && (
                <View 
                  className="flex items-center gap-1 text-purple-600"
                  onClick={() => setShowAddSheet(type)}
                >
                  <Plus size={14} />
                  <Text className="text-xs">添加</Text>
                </View>
              )}
            </View>

            {list.map(method => (
              <Card key={method.id} className="mb-2">
                <CardContent className="p-3">
                  <View className="flex items-center gap-3">
                    <Text className="text-2xl">{method.icon}</Text>
                    <View className="flex-1">
                      <View className="flex items-center gap-2">
                        <Text className="text-sm font-medium text-gray-900">{method.label}</Text>
                        {method.isDefault && (
                          <Badge className="text-xs bg-green-50 text-green-600 border-0">默认</Badge>
                        )}
                      </View>
                      <Text className="text-xs text-gray-500">{method.detail}</Text>
                    </View>

                    {/* 设为默认 */}
                    {!method.isDefault && (
                      <View 
                        className="px-3 py-1 bg-purple-50 rounded-full"
                        onClick={() => setDefault(method.id)}
                      >
                        <Text className="text-xs text-purple-600">设为默认</Text>
                      </View>
                    )}

                    {/* 删除（微信支付不可删除） */}
                    {type !== 'wechat' && (
                      <View onClick={() => removeMethod(method.id)}>
                        <Trash2 size={16} className="text-gray-400" />
                      </View>
                    )}
                  </View>
                </CardContent>
              </Card>
            ))}

            {list.length === 0 && type !== 'wechat' && (
              <Card>
                <CardContent className="p-4 text-center">
                  <Text className="text-gray-400 text-sm">暂未绑定{config.title}</Text>
                </CardContent>
              </Card>
            )}
          </View>
        )
      })}

      {/* 安全提示 */}
      <View className="px-4 mb-4">
        <View className="flex items-center gap-2 px-3 py-2 bg-blue-50 rounded-lg">
          <Shield size={14} className="text-blue-500" />
          <Text className="text-xs text-blue-700">您的支付信息已加密保护，邑夏不会存储完整卡号</Text>
        </View>
      </View>

      {/* 添加银行卡弹层 */}

        <View className="fixed inset-0 bg-black/50 z-50 flex items-end" onClick={() => setShowAddSheet('none')}>
          <View 
            className="w-full bg-white rounded-t-2xl p-6 pb-safe"

            <Text className="text-sm text-gray-500 mb-2">选择银行</Text>
            <View className="flex flex-wrap gap-2 mb-4">
              {BANK_LIST.map(bank => (
                <View
                  key={bank}
                  className={`px-3 py-1.5 rounded-full text-xs ${
                    bankForm.bankName === bank 
                      ? 'bg-purple-600 text-white' 
                      : 'bg-gray-100 text-gray-700'
                  }`}
                  onClick={() => setBankForm(prev => ({ ...prev, bankName: bank }))}
                >
                  <Text>{bank}</Text>
                </View>
              ))}
            </View>

            <View className="mb-3">
              <Text className="text-sm text-gray-500 mb-1">卡号</Text>
              <Input
                className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm"
                placeholder="请输入银行卡号"
                type="number"
                value={bankForm.cardNo}
                onInput={e => setBankForm(prev => ({ ...prev, cardNo: e.detail.value }))}
              />
            </View>

            <View className="mb-3">
              <Text className="text-sm text-gray-500 mb-1">持卡人姓名</Text>
              <Input
                className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm"
                placeholder="请输入持卡人姓名"
                value={bankForm.holderName}
                onInput={e => setBankForm(prev => ({ ...prev, holderName: e.detail.value }))}
              />
            </View>

            <View className="flex gap-3 mb-4">
              <View 
                className={`flex-1 py-2 rounded-lg text-center text-sm ${
                  bankForm.cardType === 'debit' ? 'bg-purple-600 text-white' : 'bg-gray-100 text-gray-700'
                }`}
                onClick={() => setBankForm(prev => ({ ...prev, cardType: 'debit' }))}
              >
                <Text>储蓄卡</Text>
              </View>
              <View 
                className={`flex-1 py-2 rounded-lg text-center text-sm ${
                  bankForm.cardType === 'credit' ? 'bg-purple-600 text-white' : 'bg-gray-100 text-gray-700'
                }`}

                <Text>信用卡</Text>
              </View>
            </View>

            <Button className="w-full" onClick={handleAddBankCard}>
              确认绑定
            </Button>
          </View>
        </View>
      )}

      {/* 添加支付宝弹层 */}
      {showAddSheet === 'alipay' && (
        <View className="fixed inset-0 bg-black/50 z-50 flex items-end" onClick={() => setShowAddSheet('none')}>
          <View 
            className="w-full bg-white rounded-t-2xl p-6 pb-safe"
            onClick={e => e.stopPropagation()}
          >
            <Text className="text-lg font-semibold text-gray-900 mb-4">绑定支付宝</Text>

            <View className="mb-3">
              <Text className="text-sm text-gray-500 mb-1">支付宝账号</Text>
              <Input
                className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm"
                placeholder="手机号或邮箱"
                value={alipayForm.account}
                onInput={e => setAlipayForm(prev => ({ ...prev, account: e.detail.value }))}
              />
            </View>

            <View className="mb-4">
              <Text className="text-sm text-gray-500 mb-1">实名姓名</Text>
              <Input
                className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm"
                placeholder="请输入实名认证姓名"
                value={alipayForm.realName}
                onInput={e => setAlipayForm(prev => ({ ...prev, realName: e.detail.value }))}
              />
            </View>

            <Button className="w-full" onClick={handleAddAlipay}>
              确认绑定
            </Button>
          </View>
        </View>
      )}
    </View>
  )
}

/**
 * 充值卡状态管理
 * 充值卡绑定、余额查询、扣款记录

import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import taroStorage from './taroStorage'

// 充值卡
export interface RechargeCard {
  id: string
  cardNo: string          // 卡号
  balance: number         // 余额
  initialAmount: number   // 初始金额
  discount: number        // 折扣率（0.9=9折）
  source: string          // 来源描述
  expireAt: string        // 过期时间
  isDefault: boolean
}

// 扣款记录
export interface RechargeRecord {
  id: string
  cardId: string
  type: 'topup' | 'spend' | 'refund'
  amount: number
  balance: number
  description: string
  createdAt: string
}

interface RechargeState {
  cards: RechargeCard[]
  records: RechargeRecord[]

  // 操作
  bindCard: (cardNo: string, amount: number, discount: number, source: string, expireAt: string) => void
  removeCard: (id: string) => void
  setDefault: (id: string) => void
  deduct: (cardId: string, amount: number, description: string) => boolean  // 扣款，返回是否成功
  getTotalBalance: () => number
}

const generateId = () => `rc_${Date.now()}_${Math.random().toString(36).substr(2, 6)}`

export const useRechargeStore = create<RechargeState>()(
  persist(
    (set, get) => ({
      cards: [
        {
          id: 'rc_init_1',
          cardNo: 'YX2026****8888',
          balance: 350,
          initialAmount: 500,
          discount: 0.9,
          source: '邑夏开学季充值卡',
          expireAt: '2027-03-01',
          isDefault: true
        }
      ],
      records: [
        { id: 'rr_1', cardId: 'rc_init_1', type: 'topup', amount: 500, balance: 500, description: '充值', createdAt: '2026-03-01 10:00' },
        { id: 'rr_2', cardId: 'rc_init_1', type: 'spend', amount: 85, balance: 415, description: '购买蜜桃果酒x2', createdAt: '2026-04-05 14:30' },
        { id: 'rr_3', cardId: 'rc_init_1', type: 'spend', amount: 65, balance: 350, description: '购买石榴果酒x1+配送', createdAt: '2026-05-10 20:15' }
      ],

      bindCard: (cardNo, amount, discount, source, expireAt) => {
        const newCard: RechargeCard = {
          id: generateId(),
          cardNo,
          balance: amount,
          initialAmount: amount,
          discount,
          source,
          expireAt,
          isDefault: get().cards.length === 0
        }
        const record: RechargeRecord = {
          id: generateId(),
          cardId: newCard.id,
          type: 'topup',
          amount,
          balance: amount,
          description: `绑定充值卡 ${cardNo}`,
          createdAt: new Date().toLocaleString('zh-CN')
        }
        set(state => ({
          cards: [...state.cards, newCard],
          records: [record, ...state.records]
        }))
      },

      removeCard: (id) => {
        set(state => {
          const cards = state.cards.filter(c => c.id !== id)
          if (state.cards.find(c => c.id === id)?.isDefault && cards.length > 0) {
            cards[0].isDefault = true
          }
          return { cards }
        })
      },

      setDefault: (id) => {
        set(state => ({
          cards: state.cards.map(c => ({ ...c, isDefault: c.id === id }))
        }))
      },

      deduct: (cardId, amount, description) => {
        const card = get().cards.find(c => c.id === cardId)
        if (!card || card.balance < amount) return false
        
        const discountedAmount = Math.round(amount * card.discount * 100) / 100
        set(state => ({
          cards: state.cards.map(c => 
            c.id === cardId ? { ...c, balance: Math.round((c.balance - discountedAmount) * 100) / 100 } : c
          ),
          records: [
            {
              id: generateId(),
              cardId,
              type: 'spend',
              amount: discountedAmount,
              balance: Math.round((card.balance - discountedAmount) * 100) / 100,
              description: `${description}（${(card.discount * 10).toFixed(0)}折）`,
              createdAt: new Date().toLocaleString('zh-CN')
            },
            ...state.records
          ]
        }))
        return true
      },

      getTotalBalance: () => {
        return get().cards.reduce((sum, c) => sum + c.balance, 0)
      }
    }),
    {
      name: 'yixia-recharge-storage',
      partialize: (state) => ({
        cards: state.cards,
        records: state.records
      })
    }
  )
)

        <Card className="mb-4">
          <CardContent className="p-4">
            <Text className="text-sm font-semibold text-gray-900 mb-3">提现到</Text>
            {methods.map(method => (
              <View
                key={method.id}
                className={`flex items-center gap-3 p-3 rounded-xl mb-2 ${
                  selectedMethodId === method.id ? 'bg-purple-50 border-2 border-purple-300' : 'bg-gray-50 border-2 border-transparent'
                }`}
                onClick={() => setSelectedMethodId(method.id)}
              >
                <Text className="text-2xl">{method.icon}</Text>
                <View className="flex-1">
                  <Text className="text-sm font-medium text-gray-900">{method.label}</Text>
                  <Text className="text-xs text-gray-500">{method.detail}</Text>
                </View>
                {selectedMethodId === method.id && (
                  <CheckCircle size={20} className="text-purple-600" />
                )}
              </View>
            ))}
          </CardContent>
        </Card>

        {/* 提现按钮 */}
        <Button 
          className="w-full mb-4" 
          size="lg"
          disabled={!amount || parseFloat(amount) <= 0 || parseFloat(amount) > availableBalance}
          onClick={handleWithdraw}
        >
          确认提现 ¥{amount || '0.00'}
        </Button>

        {/* 提示 */}
        <View className="flex items-start gap-2 px-3 py-2 bg-blue-50 rounded-lg mb-4">
          <AlertCircle size={14} className="text-blue-500 mt-0.5" />
          <View className="text-xs text-blue-700">
            <Text>• 提现到微信：实时到账</Text>
            {'\n'}
            <Text>• 提现到银行卡/支付宝：1-3个工作日</Text>
            {'\n'}
            <Text>• 最低提现金额 ¥10，每日限额 ¥2000</Text>
          </View>
        </View>

        {/* 提现记录 */}
        <Text className="text-sm font-semibold text-gray-900 mb-2">提现记录</Text>
        {MOCK_RECORDS.map(record => {
          const config = statusConfig[record.status]
          return (
            <Card key={record.id} className="mb-2">
              <CardContent className="p-3 flex items-center justify-between">
                <View>
                  <Text className="text-sm text-gray-900">提现到{record.method}</Text>
                  <Text className="text-xs text-gray-400">{record.methodDetail} · {record.createdAt}</Text>
                </View>
                <View className="text-right">
                  <Text className="text-sm font-bold text-gray-900">-¥{record.amount.toFixed(2)}</Text>
                  <Text className={`text-xs ${config.color}`}>{config.label}</Text>
                </View>
              </CardContent>
            </Card>
          )
        })}
      </View>
    </View>
  )
}
```

## 数据流关系图

```
用户下单 → 选择支付方式
  ├── 充值卡优先（9折）→ rechargeStore.deduct()
  ├── 微信支付 → 调用微信支付API
  ├── 支付宝 → 调用支付宝SDK
  └── 银行卡 → 调用银联支付SDK

支付成功 → 
  ├── 积分获取 → pointsStore.earnPoints()
  ├── 订单状态更新 → paid
  └── 精灵碎片发放 → spriteStore

跑腿员接单 → 完成配送 → 
  ├── 跑腿员收入增加
  └── 提现 → withdraw页面 → paymentStore选择的提现方式
```

## 注意事项

1. **微信支付绑定**：微信支付自动绑定当前微信账号，无需手动添加，也不可删除
2. **充值卡9折**：充值卡消费默认享9折，扣款时自动计算折扣金额
3. **积分对话**：当前为本地规则匹配回复，后续可接入AI对话
4. **跑腿员中心**：仅 agent 角色可进入，需在页面入口做权限校验
5. **提现**：跑腿员提现只能提到已绑定的支付方式，需至少绑定一种
6. **Mock数据**：所有页面含Mock数据，可直接预览效果，生产环境替换为API调用
